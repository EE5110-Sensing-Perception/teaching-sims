\begin{frame}{Why inertial sensing matters in robotics}
  \begin{itemize}
    \item Every mobile robot carries an IMU: drones, legged robots, AGVs, manipulators on mobile bases.
    \item It is the \textbf{only} sensor that is self-contained, high-rate (\SIrange{100}{2000}{\hertz}) and never ``loses track''.
    \item But it measures \emph{derivatives} (rate, specific force), so errors integrate into drift.
    \item The rest of the stack (VIO, LIO, GNSS/INS, legged-state estimation) exists to bound that drift.
  \end{itemize}
  \vfill
  \begin{block}{Theme of the module}
    Know what the sensor measures, how it is wrong, and how to fuse it so the errors stay bounded.
  \end{block}
\end{frame}

\begin{frame}{Outline (1/2)}
  \small
  \textbf{1. Frames and rotations}
  \begin{itemize}
    \item Body, navigation and sensor frames; aerospace vs ROS conventions
    \item Direction cosine matrix (DCM), Euler angles, quaternion, axis-angle; composition order; gimbal lock
    \item Attitude kinematics and integration
  \end{itemize}
  \textbf{2. Inertial sensors}
  \begin{itemize}
    \item MEMS accelerometer, specific force and static tilt
    \item Coriolis vibratory gyroscope
    \item Magnetometer heading
  \end{itemize}
  \textbf{3. Sensor errors and calibration}
  \begin{itemize}
    \item Bias, scale factor, misalignment; noise and drift
    \item Allan variance; datasheet to model
    \item Six-position and magnetometer calibration
  \end{itemize}
\end{frame}

  \textbf{5. Inertial navigation}

    \item Error growth: $t$, $t^2$, $t^3$

  \begin{itemize}

    \item Lever arm and mounting
  \end{itemize}
\end{frame}
