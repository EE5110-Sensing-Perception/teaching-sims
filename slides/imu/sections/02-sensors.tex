\begin{frame}{MEMS accelerometer: a proof mass on springs \hfill \lp{6}}

      \begin{tikzpicture}[>=Stealth, thick, scale=0.95]

        \draw (0.3,-1.2) -- (0.3,1.2);
        \draw[decorate,decoration={zigzag,segment length=5pt,amplitude=3pt}] (0.3,0.5) -- (2,0.5);

        \draw (1.0,-0.75) -- (1.0,-0.25);
        \draw (0.9,-0.75) -- (1.3,-0.75) -- (1.3,-0.25) -- (0.9,-0.25);
        \draw (1.3,-0.5) -- (2,-0.5);
        \node[below] at (1.15,-0.8) {\scriptsize $c$};
        \draw[fill=MediumBlue!25] (2,-0.9) rectangle (3.2,0.9);
        \node at (2.6,0) {$m$};
        \draw[->] (2.6,1.1) -- (3.3,1.1) node[right] {\scriptsize $x$};
        \draw[->, maroon] (-0.4,-1.5) -- (0.6,-1.5) node[right] {\scriptsize $a_{\mathrm{ext}}$ (case)};
        \draw (3.45,-0.9) -- (3.45,0.9);
        \draw (3.65,-0.9) -- (3.65,0.9);
        \node[right] at (3.7,0) {\scriptsize $C(x)$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}
      \small
      \[ m\ddot x + c\dot x + kx = -m\,a_{\mathrm{ext}} \]
      \[ \frac{X}{A}(s) = \frac{-1}{s^2 + 2\zeta\omega_0 s + \omega_0^2},\quad \omega_0=\sqrt{k/m} \]
      \begin{itemize}
        \item Static: $x = -a/\omega_0^2$, which is \SI{49}{\nano\metre} per $g$ at $f_0=\SI{2.25}{\kilo\hertz}$.
        \item Differential comb: $\Delta C \approx 2C_0\,x/g_0$ (linear, and it rejects common-mode).
        \item Bias at rest: an electronics offset, \emph{or} a mechanical rest offset multiplied by $\omega_0^2$.
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{Sensitivity, bandwidth and noise: pick two \hfill \lp{6}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\linewidth]{mems_accel_response.pdf}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      \begin{itemize}
        \item Soft spring: more motion per $g$, but lower $f_0$ and bandwidth.
        \item Light damping: vibration near $f_0$ is \emph{amplified}.
        \item Brownian floor $a_n = \sqrt{4k_BT\omega_0/(mQ)}$: a bigger mass means less noise.
        \item Above the bandwidth, vibration is under-reported. Above Nyquist it \emph{aliases}.
      \end{itemize}
    \end{column}
  \end{columns}
  \tryit{mems-accel}{soft\_vs\_stiff, vibration\_above\_bw, impulse\_ring, overdamped\_impulse}
\end{frame}

\begin{frame}{Accelerometers measure specific force \hfill \lp{7}}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \[ \vect{f}^b = \mat{R}^{b}_{n}\,(\vect{a}^n - \vect{g}^n) \]
      \begin{itemize}
        \item Free fall: $\vect{f}=0$. At rest: $\vect{f}=-\vect{g}$ (up), i.e.\ $[0,0,-g]$ level in FRD.
        \item Static tilt from the gravity direction:
          \[ \phi=\operatorname{atan2}(-f_y,-f_z) \]
          \[ \theta=\operatorname{atan2}\!\big(f_x,\sqrt{f_y^2+f_z^2}\big) \]
        \item \textbf{Yaw is unobservable}: gravity is invariant to rotation about the vertical.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{accel_disturbance_tilt.pdf}

      \small A static accelerometer cannot tell a tilt $\delta\theta$ from a bias $b_x$, a forward acceleration $a_x$, or a lever-arm term. Each looks like $\approx g\,\delta\theta$.
    \end{column}
  \end{columns}
  \tryit{accelerometer}{level\_plate, static\_tilt, bias\_tilts\_estimate, surge\_contaminates}
\end{frame}

\begin{frame}{Vibration and averaging \hfill \lp{7}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\linewidth]{accel_vibration_average.pdf}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      \begin{itemize}
        \item Zero-mean vibration and noise average out: the error falls as $1/\sqrt{N}$.
        \item Bias, sustained acceleration and lever-arm terms do \textbf{not} average out.
        \item Large pitch: the horizontal axes lose sensitivity to roll, and near $\pm90^\circ$ the formula becomes fragile.
        \item $\Rightarrow$ gravity is a good \emph{low-frequency} reference for tilt. We use this in the complementary filter.
      \end{itemize}
    \end{column}
  \end{columns}
  \tryit{accelerometer}{vibration\_average, large\_pitch}
\end{frame}

\begin{frame}{Coriolis vibratory gyroscope \hfill \lp{8}}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[>=Stealth, thick, scale=0.9]
        \draw[gray, rounded corners] (-2,-1.6) rectangle (2,1.6);
        \draw[fill=MediumBlue!25] (-0.6,-0.45) rectangle (0.6,0.45);
        \node at (0,0) {\scriptsize $m$};
        \draw[<->, MediumBlue] (-1.5,0) -- (-0.7,0) node[midway,above] {\scriptsize drive $x$};
        \draw[->, maroon] (0,0.5) -- (0,1.3) node[above] {\scriptsize $F_c = -2m\,\boldsymbol{\Omega}\times\vect{v}$};
        \draw[->, MediumBlue] (0.65,-0.2) -- (1.5,-0.2) node[below] {\scriptsize $\vect{v}$};
        \draw[->, orange] (1.75,-0.9) arc (0:300:0.25);
        \node[orange] at (1.5,-1.35) {\scriptsize $\Omega_z$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}
        \item The drive mode is held at resonance by an AGC: $x=A\sin\omega_d t$.
        \item Rotation couples it into the sense mode:
          \[ \ddot y + \tfrac{\omega_y}{Q}\dot y + \omega_y^2 y = -2\Omega\,\dot x - \tfrac{k_{xy}}{m}x \]
        \item Coriolis is in phase with \textbf{velocity}; quadrature ($k_{xy}$) is in phase with \textbf{displacement}.
        \item Synchronous demodulation at $\omega_d$ followed by a low-pass filter gives $\Omega$.
      \end{itemize}
    \end{column}
  \end{columns}
  \tryit{mems-gyro}{no\_rotation, coriolis\_step, quadrature\_rejected}
\end{frame}

\begin{frame}{Where gyro bias comes from: quadrature leakage \hfill \lp{8}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{mems_gyro_quadrature.pdf}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      \begin{itemize}
        \item Quadrature is often \emph{hundreds} of deg/s-equivalent, set by etching tolerances.
        \item With a demodulator phase error $\varphi$:
          \[ \hat\Omega = \Omega\cos\varphi - \Omega_q\sin\varphi \]
        \item The phase drifts with temperature, so the bias drifts too. This is a root cause of turn-on bias and in-run instability.
        \item Mitigations: quadrature-nulling electrodes, phase calibration, and temperature compensation.
      \end{itemize}
    \end{column}
  \end{columns}
  \tryit{mems-gyro}{quadrature\_bias}
\end{frame}

\begin{frame}{Mode matching: sensitivity vs bandwidth \hfill \lp{8}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\linewidth]{mems_gyro_modes.pdf}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      \begin{itemize}
        \item Matched ($f_y=f_d$): about $Q\times$ the motion, so lower noise in deg/s, but the bandwidth is only $\approx f_d/2Q$.
        \item Split by $\Delta f$: a flat response up to $\sim\Delta f$, with lower sensitivity.
        \item Consumer MEMS gyros usually run split mode. Navigation-grade MEMS push towards matched mode with closed-loop control.
      \end{itemize}
    \end{column}
  \end{columns}
  \tryit{mems-gyro}{mode\_matched, mode\_split, sine\_rate\_bandwidth}
\end{frame}

\begin{frame}{Magnetometer heading \hfill \lp{9}}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      Earth field in NED, from intensity $F$, dip $I$ and declination $D$:
      \[ \vect{b}^n = F\,[\cos I\cos D,\ \cos I\sin D,\ \sin I] \]
      \begin{itemize}
        \item Level: $\psi_m=\operatorname{atan2}(-b_y,\,b_x)$.
        \item Tilted: rotate $\vect{b}^b$ back to level with accel roll/pitch (\emph{tilt compensation}).
        \item Dip $\approx65^\circ$: vertical $\approx2\times$ horizontal, so a small pitch error means a large heading error.
        \item True heading $=\psi_m + D$ (declination from the WMM).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{mag_tilt_error.pdf}
    \end{column}
  \end{columns}
  \tryit{magnetometer}{pitched\_needs\_tc, dip\_side\_view, declination}
\end{frame}

\begin{frame}{Magnetic disturbances: detect and gate \hfill \lp{9}}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Body-fixed} (hard and soft iron on the robot): calibratable, see the next section.
        \item \textbf{World-fixed} (steel floors, rebar, parked cars): not calibratable, and they vary with position.
        \item \textbf{Time-varying} (motor currents, actuators): separate the magnetometer physically, or model it against current.
      \end{itemize}
      Consistency checks before using a sample:
      \[ \big|\,\|\vect b\| - F\,\big| < 5\%,\qquad |\hat I - I| < 3^\circ \]
      If a check fails, skip the magnetometer update and let the gyro coast.
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{alertblock}{Robotics reality}
        \small Indoors, magnetometer heading is often unusable. Robots then rely on gyro + vision/LiDAR for yaw, which is why yaw is the hardest state to keep in VIO.
      \end{alertblock}
    \end{column}
  \end{columns}
  \tryit{magnetometer}{steel\_disturbance}
\end{frame}
