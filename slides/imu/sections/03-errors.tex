\section{Sensor errors and calibration}

\begin{frame}{Deterministic errors}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      Measurement model for each triad (accelerometer or gyro):
      \[
        \tilde{\vect{y}} = (\mat{I}+\mat{S})\,\mat{M}\,\vect{y} + \vect{b} + \vect{n}
      \]
      \vspace{-1em}
      \begin{itemize}
        \item $\vect{b}$ \textbf{bias}: turn-on (changes every power cycle) plus in-run (drifts).
        \item $\mat{S}$ \textbf{scale factor}: $\operatorname{diag}(s_x,s_y,s_z)$, in ppm or \%.
        \item $\mat{M}$ \textbf{misalignment / non-orthogonality}: small off-diagonal terms, in mrad.
        \item Also: gyro g-sensitivity, temperature, non-linearity.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{block}{Where they come from}
        \begin{itemize}
          \item Bias: electronics offset, mechanical rest offset, gyro quadrature leakage.
          \item Scale: spring stiffness, gap, and gain tolerances.
          \item Misalignment: die placement, package and PCB mounting.
        \end{itemize}
      \end{block}
      \vspace{0.4em}
      \begin{alertblock}{Key point}
        Scale and misalignment errors grow with the signal. A level accelerometer hides them. Tilt it and they appear.
      \end{alertblock}
    \end{column}
  \end{columns}
  \examplebar{accelerometer}{scale\_misalignment}
\end{frame}

\begin{frame}{Allan variance: the IMU fingerprint}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \small
      \begin{enumerate}
        \item Record at rest for hours; average in clusters of length $\tau$.
        \item $\sigma_A^2(\tau) = \tfrac12\big\langle(\bar\omega_{k+1}-\bar\omega_k)^2\big\rangle$
        \item Read the slopes on a log-log plot:
      \end{enumerate}
      \begin{itemize}
        \item $-\tfrac12$: white noise, $N=\sigma_A(\SI{1}{s})$
        \item $0$: bias instability, floor $=0.664\,B$
        \item $+\tfrac12$: rate random walk, $K=\sigma_A(\SI{3}{s})$ on that line
      \end{itemize}
      \footnotesize Trust only $\tau \lesssim T/10$. A constant bias is invisible to Allan.
    \end{column}
    \begin{column}{0.56\textwidth}
      \includegraphics[width=\linewidth]{allan_signatures.pdf}
    \end{column}
  \end{columns}
  \examplebar{gyroscope}{allan\_industrial}
\end{frame}

\begin{frame}{Six-position accelerometer calibration}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \begin{enumerate}
        \item Six static poses, each axis up and then down: $\vect{f}_k=\pm g\,\vect{e}_k$.
        \item Average $N$ samples per pose.
        \item Least squares per sensor axis $i$ (6 equations, 4 unknowns):
          \[ \bar y_{k,i} = \vect{G}_i^\top \vect{f}_k + b_i \]
        \item Apply $\hat{\vect{f}} = \hat{\mat{G}}^{-1}(\tilde{\vect{y}}-\hat{\vect{b}})$.
      \end{enumerate}
      \footnotesize Gyros: bias from a rest average. Scale needs a rate table.
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{accel_six_position.pdf}
    \end{column}
  \end{columns}
  \examplebar{accelerometer}{six\_position\_cal}
\end{frame}

\begin{frame}{Magnetometer calibration: the compass swing}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \small
      \[ \tilde{\vect b} = \mat{S}\,\vect b + \vect h \]
      \begin{itemize}
        \item Hard iron shifts the locus; soft iron makes it an ellipse.
        \item Swing $360^\circ$ \emph{level}; least-squares fit of $ax^2{+}bxy{+}cy^2{+}dx{+}ey{=}1$.
        \item Correct: $\mat W(\tilde{\vect b}-\vect c)$ maps the ellipse onto a circle.
        \item 3-D ellipsoid fit: tumble through all orientations.
      \end{itemize}
    \end{column}
