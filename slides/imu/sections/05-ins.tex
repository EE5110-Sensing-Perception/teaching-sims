\begin{frame}{Strapdown mechanisation}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \centering
      \begin{tikzpicture}[>=Stealth, thick, node distance=0.35cm and 0.45cm,
          blk/.style={draw, rounded corners, minimum height=0.75cm, minimum width=1.3cm, font=\scriptsize, align=center}]
        \node[blk, draw=MediumBlue] (gy) {gyro\\$\tilde{\boldsymbol\omega}$};
        \node[blk, draw=MediumBlue, right=of gy] (att) {attitude\\$\dot{\mat R}=\mat R\skewm{\boldsymbol\omega}$};
        \node[blk, draw=MediumGreen, below=0.6cm of gy] (ac) {accel\\$\tilde{\vect f}^b$};
        \node[blk, draw=maroon, right=of ac] (rot) {$\mat R\,\tilde{\vect f}^b$};
        \node[blk, draw=maroon, right=of rot] (g) {$+\,\vect g^n$};
        \node[blk, draw=maroon, right=of g] (v) {$\int$ $\vect v^n$};
        \node[blk, draw=maroon, below=0.5cm of v] (p) {$\int$ $\vect p^n$};
        \draw[->] (gy) -- (att);
        \draw[->] (att) -- (rot);
        \draw[->] (ac) -- (rot);
        \draw[->] (rot) -- (g);
        \draw[->] (g) -- (v);
        \draw[->] (v) -- (p);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      \begin{itemize}
        \item Rotate specific force to nav, add gravity back, then integrate twice.
        \item Every attitude error resolves $g$ into the wrong direction: $\delta\vect a \approx -g\,\delta\theta$.
        \item \textbf{0.1$^\circ$ of tilt = 1.7 mg of accel bias.}
        \item Full mechanisation adds Earth rate, transport rate, and Coriolis terms. The vertical channel is unstable, so it needs a baro or GNSS.
      \end{itemize}
    \end{column}
  \end{columns}
  \examplebar{ins}{perfect\_circle, tilt\_leaks\_gravity}
\end{frame}

\begin{frame}{Aiding bounds the error}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \includegraphics[width=\linewidth]{ins_aiding.pdf}
    \end{column}
    \begin{column}{0.4\textwidth}
      \small
      \begin{itemize}
        \item \textbf{ZUPT}: when stationary, $\vect v=0$ is a measurement. Used every step by foot-mounted INS.
        \item The detector matters: at constant velocity the IMU reads ``nothing happening'', exactly like a stop.
        \item \textbf{Position/velocity fixes} (GNSS, odometry, VIO) in a Kalman filter turn drift into a bounded sawtooth.
        \item Error-state KF (ESKF): states $\delta\vect p, \delta\vect v, \delta\boldsymbol\theta, \vect b_a, \vect b_g$. This is the core of GNSS/INS, VIO and LIO.
      \end{itemize}
    \end{column}
  \end{columns}
  \examplebar{ins}{stop\_and\_go, zupt, zupt\_false\_detection, position\_fixes}
\end{frame}

\begin{frame}{How long can your robot coast?}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{ins_grades.pdf}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small
      \begin{itemize}
        \item Use the residual (post-calibration) biases from the datasheet: bias instability, VRW and ARW.
        \item The coast time before error exceeds your tolerance sets the minimum aiding rate.
        \item Robotics rule of thumb: consumer MEMS is fine with aiding at $\geq$ 1--10 Hz. Unaided coasting longer than about a second needs better grades or motion constraints (non-holonomic, ZUPT).
      \end{itemize}
    \end{column}
  \end{columns}
  \examplebar{ins}{grade\_consumer (switch the IMU grade)}
\end{frame}
